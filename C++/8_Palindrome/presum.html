前綴和 (Prefix Sum) 基礎實作動畫

#include <iostream>
#include <vector>

using namespace std;

int main() {
    vector<int> arr = {3, 1, 4, 1, 5, 9, 2};
    int n = arr.size();

    // 建立前綴和陣列，大小與原陣列相同
    vector<int> prefix(n);

    // 基礎情況：第一個元素的前綴和等於原陣列的第一個元素
    prefix[0] = arr[0];

    // 從索引 1 開始計算前綴和
    for (int i = 1; i < n; i++) {
        prefix[i] = prefix[i-1] + arr[i];
    }

    // 輸出觀察
    for (int i = 0; i < n; i++) {
        cout << prefix[i] << " ";
    }
    
    return 0;
}
